# Appended to the boot/recovery ramdisk (kernel CONFIG_BOOT_CONFIG_FORCE), because S-Boot
# writes its own /chosen/bootargs. S-Boot's androidboot.* (hardware=samsungexynos7870,
# serialno, bootloader, ...) still arrive on the cmdline.

androidboot.super_partition = SYSTEM
androidboot.boot_devices = soc@0/13540000.mmc
androidboot.selinux = permissive
androidboot.console = ttySAC2

# Kernel parameters (not early params: those cannot come from bootconfig).
# clk/pd_ignore_unused stay until the clock/power-domain users are complete.
kernel.s3c2410_wdt.tmr_atboot = 1
kernel.clk_ignore_unused
kernel.pd_ignore_unused
# loglevel/ignore_loglevel are early params, and S-Boot passes loglevel=4: set the console
# level as a sysctl (applied before init runs) so init/apexd messages reach pstore, and stop
# /dev/kmsg rate-limiting init's own log lines.
kernel.sysctl.kernel.printk = "8 4 1 7"
kernel.printk.devkmsg = on
# The modem's firmware image and NV data (radio HAL, modem-setup.sh): the kernel reads them
# itself (partial firmware reads have no ueventd fallback); other firmware still comes
# from ueventd's firmware_directories
kernel.firmware_class.path = "/data/vendor/modem/firmware"
